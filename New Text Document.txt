use serde::{Serialize, Deserialize};

#[derive(Serialize, Deserialize, Clone)]
pub struct AnonymousGroupMessage {
    pub message_id: String,
    pub group_id: String,
    pub timestamp: u64,
    
    // The main WebP bytes or text (Hides the sender_id inside!)
    pub encrypted_payload_bytes: Vec<u8>,
    
    // A completely anonymous list of key envelopes with no IDs attached
    pub anonymous_key_envelopes: Vec<Vec<u8>>, 
}

// Inside your receiver processing logic:
async fn try_decrypt_message(
    message: AnonymousGroupMessage, 
    my_private_pgp_key: &MyPgpKey
) -> Option<DecryptedPayload> {
    
    let mut decrypted_secret_key: Option<Vec<u8>> = None;

    // Loop through every envelope until one opens successfully
    for envelope_bytes in message.anonymous_key_envelopes {
        // Attempt PGP decryption
        if let Ok(secret_key) = decrypt_pgp_envelope(&envelope_bytes, my_private_pgp_key) {
            decrypted_secret_key = Some(secret_key);
            break; // Stop the loop immediately on success!
        }
    }

    // If we successfully opened an envelope, decrypt the main payload
    if let Some(secret_key) = decrypted_secret_key {
        if let Ok(payload) = decrypt_main_payload(&message.encrypted_payload_bytes, &secret_key) {
            // This payload struct now contains the sender_id and the actual text/media bytes!
            return Some(payload);
        }
    }

    None // The envelope was not meant for this user
}
