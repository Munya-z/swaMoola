use wasm_bindgen_futures::JsFuture;
use web_sys::{HtmlCanvasElement, HtmlImageElement, Url, Blob};

async fn convert_to_clean_webp_bytes(file: web_sys::File) -> Result<Vec<u8>, leptos::prelude::JsValue> {
    let window = web_sys::window().unwrap();
    let object_url = Url::create_object_url_with_blob(&file)?;
    let img = window.document().unwrap().create_element("img")?.dyn_into::<HtmlImageElement>()?;
    img.set_src(&object_url);

    // Wait for the browser to finish loading the image
    let (tx, rx) = futures::channel::oneshot::channel();
    let mut tx = Some(tx);
    let closure = wasm_bindgen::closure::Closure::once(move || { if let Some(t) = tx.take() { let _ = t.send(()); } });
    img.set_onload(Some(closure.as_ref().unchecked_ref()));
    let _ = rx.await; 

    // Paint to a hidden canvas (This step instantly deletes hidden metadata blocks)
    let canvas = window.document().unwrap().create_element("canvas")?.dyn_into::<HtmlCanvasElement>()?;
    canvas.set_width(img.natural_width());
    canvas.set_height(img.natural_height());
    canvas.get_context("2d")?.unwrap().dyn_into::<web_sys::CanvasRenderingContext2d>()?.draw_image_with_html_image_element(&img, 0.0, 0.0)?;
    Url::revoke_object_url(&object_url)?; 

    // Compress to small WebP bytes
    let (tx_blob, rx_blob) = futures::channel::oneshot::channel();
    let mut tx_blob = Some(tx_blob);
    let blob_closure = wasm_bindgen::closure::Closure::once(move |b: leptos::prelude::JsValue| { if let Some(t) = tx_blob.take() { let _ = t.send(b); } });
    canvas.to_blob_with_type_and_encoder_options(blob_closure.as_ref().unchecked_ref(), "image/webp", &leptos::prelude::JsValue::from_f64(0.80))?;
    
    let uint8_array = js_sys::Uint8Array::new(&JsFuture::from(rx_blob.await.unwrap().dyn_into::<Blob>()?.array_buffer()).await?);
    Ok(uint8_array.to_vec())
}

use wasm_bindgen::JsCast;
use wasm_bindgen_futures::JsFuture;
use web_sys::{AudioContext, AudioBuffer};

async fn convert_to_compressed_audio_bytes(file: web_sys::File) -> Result<Vec<u8>, leptos::prelude::JsValue> {
    let window = web_sys::window().unwrap();
    
    // 1. Read the audio file into an ArrayBuffer
    let array_buffer_promise = file.array_buffer();
    let array_buffer = JsFuture::from(array_buffer_promise).await?;
    
    // 2. Create an AudioContext to process the sound
    let audio_ctx = AudioContext::new()?;
    
    // 3. Decode the audio data (This instantly drops all metadata blocks)
    let decode_promise = audio_ctx.decode_audio_data(&array_buffer.dyn_into::<js_sys::ArrayBuffer>()?)?;
    let audio_buffer = JsFuture::from(decode_promise).await?.dyn_into::<AudioBuffer>()?;
    
    // 4. Extract raw channel data (Mono channel to save 50% more space)
    let mono_data = audio_buffer.get_channel_data(0)?;
    
    // 5. Convert the raw floating-point data into compact 16-bit WAV bytes
    let compressed_bytes = create_wav_bytes(&mono_data, audio_buffer.sample_rate());
    
    Ok(compressed_bytes)
}
